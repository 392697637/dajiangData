CREATE EXTENSION IF NOT EXISTS postgis;

```cmd
raster2pgsql -s 4326 -I -C -M -t 256x256 "E:\DEM\HENAN_4326.tif" public.gis_dem_henan_4326 > E:\DEM\gis_dem_henan_4326.sql


-- -s 4326：指定输入栅格数据的坐标参考系统（CRS）为 WGS 84
-- -I：创建一个索引
-- -C：创建一个带有空值的栅格表
-- -M：将栅格数据转换为 8 位无符号整数（1字节）
-- -t 256x256：指定栅格数据块的大小为 256x256 像素
-- -b 1:32BF：指定输入栅格数据的波段范围，这里表示将第一个波段（1）转换为 32 位浮点数（32BF）
-- "E:\DEM\HENAN_4326.tif"：输入的栅格数据文件路径
raster2pgsql -s 4326 -I -C -M -t 256x256 -b 1:32BF "E:\DEM\HENAN_4326.tif" public.gis_dem_henan_4326 > E:\DEM\gis_dem_henan_4326.sql

```

------------------------- 等高线入库 -------------------------
-------------------------DEV-------------------------
-- `gdal_contour`：GDAL 工具，**从 DEM 高程影像提取等高线**
-- `-a elevation`：等高线属性字段名`elevation`存储高程值
-- `-i 10`：等高距 **10 米**，每间隔 10 米生成一条等高线
-- 输入：河南 DEM 影像`HENAN-4326.tif`
-- 输出：shp 矢量文件`gis_dem_henan_contour.shp`，包含高程字段`elevation`
gdal_contour -a elevation -i 10 D:\MAPDATA\DEM\4326\HENAN-4326.tif D:\MAPDATA\DEM\4326\gis_dem_henan_contour.shp


-- `set PGPASSWORD`：临时设置 PG 数据库密码
-- `shp2pgsql`：shp 转 PostGIS 数据库表工具
-- `-s 4326`：坐标系 WGS84 (4326)
-- `-I`：自动创建空间索引
-- `-W UTF-8`：字符编码 UTF8
-- 把 shp 导入到库`ktd_lx_2026gis`，表名`public.gis_dem_henan_contour`
set PGPASSWORD=KtdpostSQL@2026!@#
shp2pgsql -s 4326 -I -W UTF-8 D:\MAPDATA\DEM\4326\gis_dem_henan_contour.shp public.gis_dem_henan_contour | psql -h 192.168.110.6 -p 5432 -U zhuoyi -d ktd_lx_2026gis
